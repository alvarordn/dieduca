\section{Magnitudes eléctricas}


\begin{frame}
\frametitle{Carga eléctrica}
\begin{columns}
\begin{column}{10cm}
\structure{\textbf{Carga eléctrica}}, $q(t)$: propiedad de las partículas elementales que constituyen la materia y que se manifiesta por medio de fuerzas eléctricas.

\medskip

\begin{itemize}
	\item El electrón, e$^-$, constituye la carga eléctrica más pequeña posible, representada como $q_e$.
	
	\item Se toma la carga del electrón como negativa, y positiva la del protón.
	
	\item Aplicando energía se puede conseguir que los electrones se desplacen.
\end{itemize}

\medskip

\structure{Unidad en el SI:} culombio [C] que equivale a $6.242\cdot 10^{18}$\,$q_e$.
\end{column}

\end{columns}


\end{frame}



	

\begin{frame}
\frametitle{Corriente eléctrica}

\structure{\textbf{Corriente eléctrica}}, $i(t)$: cantidad de carga que atraviesa una superficie de interés en la unidad de tiempo.

\begin{columns}
\begin{column}{6cm}
\begin{equation*}
    i(t)=\frac{dq(t)}{dt}
\end{equation*}
\end{column}
\begin{column}{6cm}
\fboxsep 1pt
\fcolorbox{green}{white}{

}
\end{column}
\end{columns}
\medskip

\begin{columns}
\begin{column}{10cm}

\begin{itemize}
	
	\item Por motivos históricos, se asocia el sentido del movimiento al de las cargas positivas (aunque el desplazamiento real en conductores metálicos es de los electrones).
	
	\item Se conoce como \textbf{conductividad eléctrica} a la capacidad de un material para permitir a los electrones circular a través de él.
	
	
\end{itemize}

\medskip

\structure{Unidad en el SI:} amperio [A].

\medskip

\end{column}

\begin{frame}
\frametitle{Voltaje (tensión, diferencia de potencial,...)}

\structure{\textbf{Diferencia de potencial}}, $v(t)$ o $u(t)$: se define como la energía, $w(t)$, necesaria para trasladar la unidad de carga entre dos puntos.

\medskip

\begin{columns}
\begin{column}{8cm}
\begin{equation*}
    u(t)=\frac{dw(t)}{dq(t)}
\end{equation*}

\medskip

\begin{itemize}
	
	\item Es necesario indicar el punto de mayor (+) y menor (-) potencial.
	
	\item Tomando como referencia de potencial un punto del circuito, la tensión entre dos puntos se calcula a partir de los potenciales respecto a dicha referencia: $v_{ab}(t)=v_a(t)-v_b(t)$.
	
\end{itemize}

\medskip

\structure{Unidad en el SI:} voltio [V].

\medskip


\end{column}

\begin{frame}
	\justifying
\frametitle{Flujo magnético}

\structure{\textbf{Ley de Faraday}}. Si el flujo magnético que atraviesa una superficie delimitada por una espira varía con el tiempo, en los bornes de dicha espira se genera una tensión (fuerza electromotriz) proporcional y de signo contrario a la variación de dicho flujo concatenado.
\begin{equation*}
    u(t)=-\frac{d \phi(t)}{dt}
\end{equation*}

\begin{itemize}
  \item El flujo puede provenir de un imán permanente, un \textbf{electroimán}, o de la intensidad que recorre la propia espira.
  \item Esta ley constituye la base de los generadores de energía eléctrica tradicionales.
\end{itemize}

\begin{frame}
\frametitle{Potencia y energía}

\begin{columns}
	\begin{column}{9cm}
		\structure{\textbf{Potencia}}, $p(t)$: cantidad de trabajo (energía) intercambiado por unidad de tiempo.
		\begin{equation*}
    		p(t)=\frac{dw(t)}{dt}=\frac{dw(t)}{dq(t)}\cdot\frac{dq(t)}{dt}=u(t)\cdot i(t)
		\end{equation*}
		Se obtiene a partir del voltaje entre dos puntos y la corriente que circula entre ellos.
		
		\structure{Unidad en el SI:} vatio [W].
	\end{column}
	\begin{column}{3cm}
		\begin{figure}[hbt]
			\epsfig{file=./figuras_tema_1/watt.eps,height=2cm}

\medskip
\medskip

\begin{columns}
	\begin{column}{9cm}

		\begin{equation*}
    		w(t)=\int^t_{-\infty}p(\tau) d\tau
		\end{equation*}

		\structure{Otras unidades muy utilizadas:} vatio-hora [Wh].

\section{Referencias de polaridad}

\begin{frame}
\frametitle{}
\justifying
\structure{\textbf{Objetivo}}: unificar los criterios de signos de las magnitudes eléctricas. Es normal asignar referencias asociadas al comportamiento habitual del componente.


\medskip

\structure{\textbf{Referencias pasivas}}: una vez definido el sentido positivo de la tensión, la intensidad entra por el terminal positivo. 
			\begin{figure}[hbt]
				\centering
				\includegraphics[height=2.5cm]{./figuras_tema_1/ref_pas_2.eps}
				\label{ref_pas}
			\end{figure}
	
\begin{equation*}
	\begin{split}
		\text{Si } p(t)&=u(t)\cdot i(t) > 0 \ \ \longrightarrow \ \ \text{Absorbe potencia}\\
		\text{Si } p(t)&=u(t)\cdot i(t) < 0 \ \ \longrightarrow \ \ \text{Cede potencia}
	\end{split}
\end{equation*}



Normalmente se tomarán referencias pasivas para las resistencias.

	

\end{frame}




\begin{frame}
	\frametitle{}

	\structure{\textbf{Referencias activas}}: una vez definido el sentido positivo de la tensión, la intensidad entra por el terminal negativo. 
	
	\begin{figure}[hbt]
		\centering
		\includegraphics[height=2.5cm]{./figuras_tema_1/ref_pas_3.eps}
		\label{ref_act}
	\end{figure}
	
	\vspace{-0.7cm}
	
	\begin{equation*}
		\begin{split}
			\text{Si } p(t)&=u(t)\cdot i(t) > 0 \ \ \longrightarrow \ \ \text{Cede potencia}\\
			\text{Si } p(t)&=u(t)\cdot i(t) < 0 \ \ \longrightarrow \ \ \text{Absorbe potencia}
		\end{split}
	\end{equation*}
	
	\justifying
	
	\medskip
	
	Normalmente se tomarán referencias activas para las fuentes de tensión e intensidad.
	
	\medskip
	
	Las referencias son solamente un criterio de signos, la realidad  no se ve afectada por utilizar unas u otras en el estudio del circuito.

\end{frame}





\section{Clasificación de circuitos}
\begin{frame}\frametitle{}
\vspace{-0.1cm}
\begin{block}{\textbf{Circuitos lineales}}
\begin{small}
	\justifying
Son circuitos en los que existe una relación lineal entre las excitaciones (tensiones e intensidades impuestas por las fuentes) y las respuestas.

\medskip

Conocidas las respuestas de un circuito ($i_1, i_2$) ante diversas excitaciones ($u_1, u_2$), la respuesta a cualquier combinación lineal de las excitaciones se corresponde con la misma combinación lineal de las respuestas.

\begin{figure}[hbt]
\centering
\fbox{\includegraphics[height=2cm]{./figuras_tema_1/propor_MAGC.png}}
\label{propor}
\end{figure}

Es consecuencia de los principios de proporcionalidad y superposición asociados a las ecuaciones lineales.

\end{small}

\end{block}
\vspace{-0.2cm}
\begin{block}{\textbf{Circuitos no lineales}}
\begin{small}
Generalmente son circuitos cuyos parámetros son función de la intensidad que lo recorre o de la tensión aplicada y  casi siempre conducen a ecuaciones no lineales.
\end{small}
\end{block}
\end{frame}



\begin{frame}\frametitle{}

\begin{block}{\textbf{Circuitos estáticos}}
	\justifying
Gobernados estrictamente por ecuaciones algebraicas.
\end{block}
\begin{block}{\textbf{Circuitos dinámicos}}
	\justifying
Gobernados por ecuaciones algebraicas y diferenciales, cuya solución en un instante dado requiere conocer la historia pasada del circuito.
\end{block}


\begin{block}{\textbf{Circuitos invariantes en el tiempo}}
	\justifying
Caracterizados por ecuaciones algebraicas o diferenciales de coeficientes constantes.
\end{block}

\begin{block}{\textbf{En función del tipo de excitación}}
	\justifying
\begin{itemize}
  \item Circuitos de \textbf{corriente continua}. Excitaciones constantes en el tiempo.
  \item Circuitos de \textbf{corriente alterna sinusoidal}. Las excitaciones son señales sinusoidales. A su vez se dividen en circuitos monofásicos y trifásicos.
\end{itemize}
\end{block}

\end{frame}

\section{Leyes de Kirchhoff}
\begin{frame}
\frametitle{Ley de Kirchhoff de corrientes}
En un nudo (corte) eléctrico donde inciden $k$ conductores cuyas corrientes se suponen entrando al nudo se cumple que
\begin{equation*}
\boxed{\sum_k i_k(t)=0} \ \ \Leftrightarrow \ \ \boxed{\sum_{i,entran} i_i(t)=\sum_{j,salen} i_j(t)}
\end{equation*}
\vspace{-0.5cm}
\begin{figure}[hbt]
\centering
\epsfig{file=./figuras_tema_1/1K_bis.eps,height=2.6cm}
\label{1K}
\end{figure}
\structure{\textcolor{red}{\textbf{Nudo}}}: interconexión de varios terminales del circuito.

\structure{\textbf{Corte}}: conjunto de interconexiones de una región del circuito con el resto.
\end{frame}


\begin{frame}
\frametitle{Ley de Kirchhoff de tensiones}
\justifying
En un bucle (malla) eléctrico formado por $k$ elementos cuyas tensiones se suponen en el mismo sentido en el que se recorre el bucle (malla) se cumple:
\begin{equation*}
\boxed{\sum_k u_k(t)=0 }
\end{equation*}

\structure{\textcolor{red}{\textbf{Bucle}}}: camino cerrado en un circuito.

\structure{\textbf{Malla}}: caso particular de bucle que no contiene ningún otro bucle en su interior.
\vspace*{-0.2cm}
\begin{figure}[hbt]
\centering
\epsfig{file=./figuras_tema_1/2K.eps,height=3cm}
\label{2K}
\end{figure}

\end{frame}

\end{exampleblock}
\end{frame}
