\begin{frame}
\frametitle{Condensador}
\begin{columns}
\begin{column}{3.5cm}

\begin{column}{8cm}
%\centering
%\hspace{2.95cm} 
\structure{\small Carga almacenada \\ (Culombios)}\hspace{\fill}$\boxed{q(t)=C\cdot u(t)}$

\vspace{0.12cm}

\structure{Relación u-i}\hspace{\fill}$\boxed{i(t)=\frac{dq(t)}{dt}=C\cdot \frac{du(t)}{dt}}$

\vspace{0.3cm}

{\small $C\equiv$ Capacidad del condensador (faradios, F)} 
%\end{center}
\end{column}
\end{columns}

  \vspace*{0.3cm}

\structure{Potencia:}\hspace{\fill}$p_C(t)=u(t)\cdot i(t)=C\cdot u(t) \cdot \frac{du(t)}{dt}=\frac{C}{2}\cdot \frac{du^2(t)}{dt}\lessgtr 0$

\vspace*{0.2cm}

\structure{Energía:}\hspace{\fill}$w_C(t)=\int^t_{-\infty} \frac{C}{2}\cdot \frac{du^2(\tau)}{d\tau}\, d\tau=\frac{C}{2}\cdot u^2(t)  \geq 0$
\mbox{}\\
\mbox{}\\

\begin{itemize}
	\item El condensador almacena energía en el campo eléctrico.
	\item La energía almacenada en cada instante depende de la tensión en el condensador.
\end{itemize}

\end{frame}

   L_1=& n^2 L_2\\
   M=& \frac{L_1}{n}=n L_2=\sqrt{L_1L_2}
   \end{split}

        \end{equation*}
siendo $n=N_1/N_2$, cociente del número de espiras en cada bobina.

\vspace{0.5cm}

El \textbf{coeficiente de acoplamiento} de dos bobinas se define como:
\begin{equation*}
    k=\sqrt{k_1\cdot k_2}=\frac{M}{\sqrt{L_1\cdot L_2}}\leq 1
\end{equation*}

Si $k=1$ el \textbf{acoplamiento es perfecto} (no existen flujos de dispersión).
\\
Si $k=0$ las bobinas están desacopladas.
