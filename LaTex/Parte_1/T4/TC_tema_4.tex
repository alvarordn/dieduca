\section[]{Balance de incógnitas y ecuaciones}

\begin{frame}\frametitle{Balance de incógnitas y ecuaciones}
	\justifying
Dado un circuito eléctrico formado por $n$ nudos y $c$ elementos de una puerta (dos terminales):
\begin{itemize}
  \item \textbf{Número de incógnitas}: $2c$ (tensión e intensidad en cada elemento)
  \item Número de ecuaciones linealmente independientes:
  \begin{itemize}
  	\justifying
  \item \textbf{Definición de los elementos}: $c$ ecuaciones.
  \item \textbf{Ley de Kirchhoff de intensidad}: $n-1$ ecuaciones.
  \item \textbf{Ley de Kirchhoff de tensión}: $c-n+1$ ecuaciones (número de mallas).
  \end{itemize}
\end{itemize}
\vspace{0.3cm}
\hrule
\begin{center}
Por tanto:\\

 \vspace{0.1cm}
 Nº Incognitas = Nº Ecuaciones\\

  \vspace{0.1cm}
 ¿Es necesario plantear $2c$ ecuaciones?\\

 $\Downarrow$\\
  \vspace{0.1cm}
 \structure{Ecuaciones de nudos ($n-1$ ecuaciones) o mallas ($c-n+1$ ecuaciones)}
\end{center}

\end{frame}

\begin{frame}\frametitle{Tensiones de nudos como variables básicas}
%\vspace{-0.2cm}

%\vspace{-0.5cm}

\vspace{-0.5cm}
\begin{figure}[hbt]
\epsfig{file=./figuras_tema_4/refer_tension.eps,height=3cm}
\end{figure}
\end{column}
\begin{column}{6cm}
$$\boxed{u_{jk}(t)=u_{N_j}(t)-u_{N_k}(t)}$$
\end{column}
\end{columns}

\vspace{0.2cm}

\begin{itemize}
	\justifying
	\item La tensión de un nudo $j$, $u_{N_j}$, se define como la diferencia de potencial entre dicho nudo y un nudo arbitrario que se toma como referencia.
	\item A partir de las tensiones nodales se puede calcular cualquier tensión del circuito y, por tanto, cualquier magnitud eléctrica.
	\item Por este motivo, se dice que las tensiones nodales son \textbf{variables básicas} de un circuito.
\end{itemize}

\end{frame}


\begin{frame}\frametitle{Ecuación de un nudo genérico}

\begin{itemize}
\justifying

\item En cada nudo se plantea la ley de Kirchhoff de intensidades, distinguiendo aquellas corrientes que provienen de fuentes de intensidad (\textbf{supuestas entrantes en el nudo}) de las que circulan por las conductancias conectadas en dicho nudo (\textbf{supuestas salientes}).

\item Estas intensidades por las conductancias se pueden expresar en función de las tensiones nodales. 

\end{itemize}

\end{figure}



\end{frame}

\begin{frame}\frametitle{Sistema de ecuaciones de nudos}

Extendiendo el balance anterior a todos los nudos del circuito (excepto el de referencia), se llega a la siguiente expresión en forma matricial:
\begin{equation*}
   {\left(
  \begin{array}{ccc}
    G_{N_{(1,1)}} & \cdots & G_{N_{(1,n-1)}} \\
    \vdots & \ddots & \vdots \\
    G_{N_{(n-1,1)}} & \cdots & G_{N_{(n-1,n-1)}}\\
  \end{array}
\right)}
    {\left(
      \begin{array}{c}
       u_{N_1}(t)  \\
        \vdots \\
        u_{N_{n-1}}(t)  \\
      \end{array}
    \right)}=
    {\left(
      \begin{array}{c}
       i_{N_1}(t)  \\
        \vdots \\
        i_{N_{n-1}}(t)  \\
      \end{array}
    \right)}
\end{equation*}
$$\Downarrow$$
$$\boxed{\textbf{G}_N\textbf{U}_N=\textbf{I}_N}$$

\vspace{0.5cm}
\begin{description}
  \item[\textbf{G}$_N$] Matriz de conductancias nodales.
  \item[\textbf{U}$_N$] Vector de tensiones de nudos.
  \item[\textbf{I}$_N$] Vector de intensidades inyectadas en los nudos.
\end{description}

\end{frame}




\begin{frame}\frametitle{Formulación por inspección}

\vspace*{-0.2cm}
\begin{small}
\begin{equation*}
   {\left(
  \begin{array}{ccc}
    G_{N_{(1,1)}} & \cdots & G_{N_{(1,n-1)}} \\
    \vdots & \ddots & \vdots \\
    G_{N_{(n-1,1)}} & \cdots & G_{N_{(n-1,n-1)}}\\
  \end{array}
\right)}
    {\left(
      \begin{array}{c}
       u_{N_1}(t)  \\
        \vdots \\
        u_{N_{n-1}}(t)  \\
      \end{array}
    \right)}=
    {\left(
      \begin{array}{c}
       i_{N_1}(t)  \\
        \vdots \\
        i_{N_{n-1}}(t)  \\
      \end{array}
    \right)}
\end{equation*}
\vspace*{-0.2cm}
\begin{enumerate}
  \item Elementos de la matriz \textbf{G}$_N$
\vspace*{-0.1cm}
\begin{equation*}
\begin{split}
G_{N_{(j,j)}}&=\sum_{k}G_{jk} \ \ \text{(suma de conductancias conectadas al nudo $j$)}\\
G_{N_{(j,k)}}&=-G_{jk} \ \ \text{(conductancia total entre el nudo $j$ y el nudo $k$)}
\end{split}
\end{equation*}
  \item Elementos del vector \textbf{I}$_N$: suma de las fuentes de intensidad que inciden en el nudo $j$, positivas si entran en el nudo y negativas si salen.
\vspace*{-0.1cm}
 \begin{equation*}
  i_{N_j}=\sum_{j} i_{N_{pj}}-i_{N_{nj}}
 \end{equation*}
\end{enumerate}
\end{small}

\end{frame}

\section[]{Ecuaciones de mallas}

\begin{frame}\frametitle{Intensidades de malla}

\begin{itemize}
  \justifying
  \item Recorre todos los elementos del circuito que conforma dicha malla.
  \item No se corresponden necesariamente con ninguna corriente eléctrica real.
  \item Se representa habitualmente mediante una flecha, ubicada en el interior de la malla respectiva, que indica el sentido que arbitrariamente se le asigna a dicha intensidad.
  \item Por conveniencia, todas las intensidades de mallas se orientan en el mismo sentido.

\begin{figure}[hbt]
\epsfig{file=./figuras_tema_4/mallas.eps,height=2cm}
\end{figure}

	\item Solo están definidas en circuitos planos, si no lo son, se pueden plantear las ecuaciones de bucles en su lugar.
	
\end{itemize}

\end{frame}

\begin{frame}\frametitle{Sistema de ecuaciones de mallas}

Extendiendo a todas las mallas del circuito:
\begin{small}
\begin{equation*}
   {\left(
  \begin{array}{ccc}
    R_{M_{(1,1)}} & \cdots & R_{M_{(1,c-n+1)}} \\
    \vdots & \ddots & \vdots \\
    R_{M_{(c-n+1,1)}} & \cdots & R_{M_{(c-n+1,c-n+1)}}\\
  \end{array}
\right)}
    {\left(
      \begin{array}{c}
       i_{M_1}(t)  \\
        \vdots \\
        i_{M_{c-n+1}}(t)  \\
      \end{array}
    \right)}=
    {\left(
      \begin{array}{c}
       u_{M_1}(t)  \\
        \vdots \\
        u_{M_{c-n+1}}(t)  \\
      \end{array}
    \right)}
\end{equation*}
\end{small}
$$\Downarrow$$
$$\boxed{\textbf{R}_M\textbf{I}_M=\textbf{U}_M}$$

\vspace{0.5cm}

\begin{description}
  \item[\textbf{R}$_M$] Matriz de resistencias de mallas.
  \item[\textbf{I}$_M$] Vector de intensidades de mallas.
  \item[\textbf{U}$_M$] Vector de tensiones aplicadas a las mallas.
\end{description}

\end{frame}


\begin{frame}\frametitle{Formulación por inspección}

\begin{small}
\begin{equation*}
   {\left(
  \begin{array}{ccc}
    R_{M_{(1,1)}} & \cdots & R_{M_{(1,c-n+1)}} \\
    \vdots & \ddots & \vdots \\
    R_{M_{(c-n+1,1)}} & \cdots & R_{M_{(c-n+1,c-n+1)}}\\
  \end{array}
\right)}
    {\left(
      \begin{array}{c}
       i_{M_1}(t)  \\
        \vdots \\
        i_{M_{c-n+1}}(t)  \\
      \end{array}
    \right)}=
    {\left(
      \begin{array}{c}
       u_{M_1}(t)  \\
        \vdots \\
        u_{M_{c-n+1}}(t)  \\
      \end{array}
    \right)}
\end{equation*}
\vspace{-0.3cm}
\begin{enumerate}
  \item Elementos de la matriz \textbf{R}$_M$
\vspace*{-0.2cm}
\begin{equation*}
\begin{split}
R_{M_{(j,j)}}&=\sum_{k}R_{jk} \ \ \text{(suma de resistencias pertenecientes a la malla $j$)}\\
R_{M_{(j,k)}}&=-R_{jk} \ \ \text{(resistencia total común de la malla $j$ y la malla $k$)}
\end{split}
\end{equation*}
  \item Elementos del vector \textbf{U}$_M$: suma de las fuentes de tensión que pertenecen a la malla $j$, con signo positivo si la intensidad de la malla sale por el terminal positivo de la fuente y signo negativo si entra.

    \begin{equation*}
	U_{M_j}=\sum_{j} u_{M_{pj}}-u_{M_{nj}} 
 \end{equation*}
\end{enumerate}
\end{small}

\end{frame}
