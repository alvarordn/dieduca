\section[]{Fuente real de tensión continua}

\begin{frame}
\frametitle{Fuente real de tensión continua}

\begin{block}{Algunas observaciones}

\begin{enumerate}
  \justifying
  \item Toda fuente real tiene una potencia máxima que puede suministrar.
  \item La tensión en bornes de una fuente real de tensión depende del circuito al que se conecte.
  \item Las fuentes reales sufren un calentamiento durante su uso.
\end{enumerate}
\end{block}

\section[]{Circuitos equivalente Thévenin y Norton}

\begin{frame}\frametitle{Circuito equivalente Thévenin}

\justifying
Todo circuito compuesto exclusivamente por fuentes y resistencias equivale a una fuente real de tensión (equivalente Thévenin).

\begin{columns}
\begin{column}{8cm}
\begin{figure}[hbt]
\epsfig{file=./figuras_tema_3/thevenin.eps,height=2.5cm}
\end{figure}
\end{column}
\begin{column}{3cm}
$$\boxed{U_{th}=\left.u_{ca}\right|_{{AB}}}$$
$$\boxed{R_{th}=\left.R_{eq}\right|_{{AB}}}$$
\end{column}
\end{columns}

\vspace{0.5cm}

El valor de la tensión de la fuente ideal es el que presenta el circuito original a intensidad nula (circuito abierto), mientras que la resistencia equivalente es la que presenta el circuito original anuladas las fuentes (circuito pasivo).

\end{frame}

\begin{frame}\frametitle{Circuito equivalente Norton}
	
\justifying
Todo circuito compuesto exclusivamente por fuentes y resistencias equivale a una fuente real de intensidad (equivalente Norton).
	
\begin{columns}
\begin{column}{8cm}
\begin{figure}[hbt]
\epsfig{file=./figuras_tema_3/norton.eps,height=2.5cm}
\end{figure}
\end{column}
\begin{column}{3cm}
$$\boxed{I_{nor}=\left.i_{cc}\right|_{{AB}}}$$
$$\boxed{R_{nor}=\left.R_{eq}\right|_{{AB}}}$$
\end{column}
\end{columns}

\vspace{0.5cm}

El valor de la corriente de la fuente ideal es el que presenta el circuito original a tensión nula (cortocircuito), mientras que la resistencia equivalente es la que presenta el circuito original anuladas las fuentes (circuito pasivo).

\end{frame}

\section{Máxima transferencia de potencia}

\begin{frame}
\frametitle{Máxima potencia de una fuente real de tensión}
\vspace{-0.5cm}
\begin{columns}
\begin{column}{5cm}
\begin{figure}[hbt]
\fboxsep 1pt
\fcolorbox{blue}{white}{\epsfig{file=./figuras_tema_3/max_pot_DC.eps,height=3cm}}
\end{figure}
\end{column}
\begin{column}{6cm}
$P=U I=R I^2=\cfrac{R}{(R+R_g)^2}\cdot U^2_g$
$$P_{max} \ \ \Longrightarrow \ \ \frac{dP}{dR}=0 \ \ \Rightarrow  \ \ \boxed{R=R_g}$$
\end{column}
\end{columns}
%\vspace{0.5cm}
%$$P_{max} \ \ \Longrightarrow \ \ \frac{dP}{dR}=0 \ \ \Rightarrow  \ \ \boxed{R=R_g}$$

%\vspace{-0.2cm}
\begin{columns}[t]
\begin{column}{6cm}
\begin{itemize}
\item Potencia máxima:
\begin{small}
  \begin{equation*}
    P_{max}=\frac{U^2_g}{4\cdot R_g}
  \end{equation*}
  \end{small}
  \vspace{-0.5cm}
 \item Rendimiento:
 \begin{small}
  \begin{equation*}
    \eta=\frac{U I}{U_g  I}=\frac{R}{R+R_g}
  \end{equation*}
  Si $R=R_g \ \ \Rightarrow \ \ \left.\eta \right|_{P_{max}}=0.5$
    \end{small}
  \end{itemize}
  \vspace{-0.8cm}
   \end{column}

  \vspace{-0.5cm}
 \begin{figure}[hbt]
