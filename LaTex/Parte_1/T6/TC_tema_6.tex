\section{Circuitos dinámicos en régimen permanente de continua}

\begin{frame}
\frametitle{Circuitos dinámicos en régimen permanente de continua}
\vspace{-0.1cm}

En régimen permanente de continua todas las magnitudes tienen un valor constante, anulándose cualquier derivada. En el caso particular de bobinas y condensadores:

%\vspace{-0.8cm}

\begin{columns}
\hspace{-0.5cm}
\begin{column}{5cm}
\begin{itemize}
	\justifying
  \item Las \textbf{bobinas} se comportan como cortocircuitos, esto es, $U=0$.
  \item Los \textbf{condensadores} se comportan como circuitos abiertos, es decir, $I=0$.
\end{itemize}
\end{column}
\hspace{-0.5cm}
\begin{column}{6cm}
\begin{figure}[hbt]
\epsfig{file=./figuras_tema_6/analisis_DC.eps,height=4cm}
\end{figure}

\end{column}
\end{columns}

\vspace{0.3cm}

\justifying
\begin{small}
	
A diferencia de un circuito estático, hay energía almacenada en bobinas y condensadores. 
Dicha energía estará involucrada en futuros transitorios cuando se produzca un cambio en el circuito. 
\end{small}

\end{frame}

\section{Análisis de circuitos dinámicos en un instante arbitrario}

\begin{frame}
\frametitle{Análisis de circuitos dinámicos en un instante arbitrario}
\justifying

Hay aplicaciones en las que interesa determinar las magnitudes de un  circuito en un instante concreto $t_0$ (condiciones iniciales de un transitorio).


\begin{enumerate}
	\justifying
  \item Las fuentes de excitación, $e_g(t)$ e $i_g(t)$, se sustituyen por fuentes de valor fijo:
  \begin{equation*}
    E_g=e_g(t_0) \ \ ; \ \ I_g=i_g(t_0)
  \end{equation*}
\item Condensadores e inductancias se sustituyen por fuentes de tensión e intensidad de valor la tensión y la intensidad en el condensador y en la inductancia, respectivamente:
  \begin{equation*}
    U_0=u_C(t_0)
%  \end{equation*}

%  \begin{equation*}
\hspace*{1cm}
    I_0=i_L(t_0)
  \end{equation*}



\end{enumerate}

%\section{Respuesta transitoria de los circuitos de primer orden}
\begin{frame}{Solución de la ecuación diferencial}

\begin{equation*}
\mathrm{RC:}\ \ \boxed{\frac{du(t)}{dt}+\frac{1}{RC}\,u(t)  = \frac{V_g}{RC}}
\hspace*{1cm}
\mathrm{RL:}\ \ \boxed{\frac{di(t)}{dt}+\frac{R}{L}\,i(t)  = \frac{R}{L}\,I_g}
\end{equation*}

La solución se compone de dos términos:
\begin{equation*}
\mathrm{RC:}\ \ \ u(t)=u_n(t)+U_\infty \hspace*{1cm}
\mathrm{RL:}\ \ \ i(t)=i_n(t)+I_\infty 
\end{equation*}
\vspace{-0.6cm}   
\begin{enumerate}
  \item \textbf{Respuesta natural:} $u_n(t)$ e $i_n(t)$
  \begin{itemize}
    \item Matemáticamente, es la solución de la ecuación diferencial homogénea.
    \item Es la respuesta del circuito cuando se anulan las fuentes. 
  \end{itemize}
  \item \textbf{Respuesta forzada:} $U_\infty$ e $I_\infty$
  \begin{itemize}
  	\justifying
    \item Depende únicamente del nuevo estado estacionario.
    \item Matemáticamente, es la solución particular de la ecuación diferencial completa .
  \end{itemize}


\end{enumerate}


\end{frame}




\begin{frame}{Respuesta natural}
\justifying
La respuesta natural se obtiene resolviendo la ecuación diferencial de primer orden homogénea:
\vspace{-0.2cm}
\begin{equation*}
\mathrm{RC:}\ \ \boxed{\frac{du(t)}{dt}+\frac{1}{\tau}\,u(t)  = 0}
\hspace*{1cm}
\mathrm{RL:}\ \ \boxed{\frac{di(t)}{dt}+\frac{1}{\tau}\,i(t)  = 0}
\end{equation*}

cuya solución, para $t\geq 0$ es

\begin{equation*}
\mathrm{RC:}\ \ \boxed{u_n(t)=K \cdot \text{e}^{-t/\tau}}
\hspace*{1cm}
\mathrm{RL:}\ \ \boxed{i_n(t)=K \cdot \text{e}^{-t/\tau}}
\end{equation*}
\vspace{-0.6cm}
\begin{itemize}
  \item $\tau$ es la constante de tiempo característica del circuito:
  \vspace{-0.2cm}
%  \begin{itemize}
%    \item 
$$
\mathrm{Circuito\  RC:}\ \ \tau=RC
\hspace*{1cm}
%    \item 
\mathrm{Circuito\  RL:}\ \ \tau=L/R
\vspace{-0.2cm}
$$
%  \end{itemize}
donde $R$ es la resistencia equivalente en bornas de $L$ o $C$.
     \item $K$ se determina a partir de las condiciones iniciales, $u(0)$ e $i(0)$
\end{itemize}

\end{frame}
