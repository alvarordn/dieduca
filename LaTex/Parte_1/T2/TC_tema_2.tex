\section[]{Resistencia y conductancia. Ley de Ohm}

\begin{frame}\frametitle{Resistencia y conductancia. Ley de Ohm}
\begin{columns}
\begin{column}{4cm}
\vspace*{-0.3cm}
\begin{figure}[hbt]
\fboxsep 5pt
\fcolorbox{blue}{white}{\epsfig{file=./figuras_tema_2/R.eps,height=2.5cm}}
\label{R}
\end{figure}
\end{column}
\begin{column}{8cm}
\centering
\structure{Relación $u(t)\leftrightarrow i(t)$}
%\vspace*{-0.3cm}
\begin{columns}
\begin{column}{4cm}
\begin{equation*}
  \boxed{u(t)=R\cdot i(t)}
\end{equation*}
\begin{flushleft}
$R\equiv$ Resistencia.\\
\structure{Unidad SI:} Ohmio [$\Omega$]
\end{flushleft}
\end{column}
\begin{column}{4cm}
\begin{equation*}
  \boxed{i(t)=G\cdot u(t)}
\end{equation*}
\begin{flushleft}
$G\triangleq \frac{1}{R}\equiv$ Conductancia.\\
\structure{Unidad SI:} Siemens [S]\\
\end{flushleft}
\end{column}
\end{columns}
\end{column}
\end{columns}


\vspace*{0.5cm}

\structure{\textbf{Ley de Ohm}:}\ para un conjunto de materiales (óhmicos) el valor de $R$ (y por tanto de $G$) no depende de la tensión ni la intensidad en la resistencia.

\vspace*{0.5cm}

\structure{\textbf{Potencia}:}\ \ $p_R(t)=u(t)\cdot i(t)=R\cdot i^2(t)=G\cdot u^2(t)\geq 0$ (siempre absorbe)

\vspace*{0.2cm}

\structure{\textbf{Energía}:}\ \ $w_R(t)=\int^t_{-\infty}R\cdot i^2(\tau)d\tau=\int^t_{-\infty}G\cdot u^2(\tau)d\tau\geq 0$
\\
\vspace*{0.2cm}
Toda la energía que absorbe del circuito la disipa en forma de calor por efecto Joule.

\section{Simplificación de circuitos resistivos}

\begin{frame}
\frametitle{Asociación serie}
\vspace*{-0.5cm}
Dos elementos están \textbf{en serie} cuando son recorridos por la misma intensidad.

%\alert{Dos elementos están en serie cuando los atraviesa la misma intensidad.}
\begin{block}{Resistencias}
\vspace*{-0.5cm}
\begin{columns}
\begin{column}{8cm}
\begin{figure}[hbt]
\epsfig{file=./figuras_tema_2/serie_R.eps,height=1.5cm}
\label{serie_R}
\end{figure}
\end{column}
\begin{column}{3cm}
$$\boxed{R_{eq}=\sum^n_{j=1} R_j}$$
\end{column}
\end{columns}
\structure{\textbf{Divisor de tensión}:} conocida $u(t)$ se puede determinar la caída de tensión en uno de los elementos, $u_k(t)$.
\begin{equation*}
    u_k(t)=R_k\cdot i(t)=R_k\cdot \frac{u(t)}{R_{eq}} \ \ \rightsquigarrow \ \  \boxed{u_k(t)=\frac{R_k}{\sum^n_{j=1} R_j}\cdot u(t)}
\end{equation*}
Es consecuencia de que todas las resistencias son atravesadas por la misma intensidad.
\end{block}
\end{frame}


\begin{frame}
\frametitle{Asociación paralelo}

Dos elementos están \textbf{en paralelo} cuando están sometidos a  la misma tensión.
\begin{block}{Resistencias}

\vspace*{-0.5cm}
\begin{columns}
\begin{column}{8cm}
\begin{figure}[hbt]
\epsfig{file=./figuras_tema_2/paralelo_R.eps,height=2cm}
\label{paralelo_R}
\end{figure}
\end{column}
\begin{column}{3cm}
	\vspace{0.1cm}
$$\boxed{G_{eq}=\sum^n_{j=1} G_j}$$
\vspace{-0.3cm}
$$\boxed{\frac{1}{R_{eq}}=\sum^n_{j=1} \frac{1}{R_{j}}}$$
\end{column}
\end{columns}
\vspace*{0.1cm}
\structure{\textbf{Divisor de intensidad}:} conocida $i(t)$ se puede determinar la intensidad en uno de los elementos, $i_k(t)$.
\vspace*{-0.4cm}
\begin{equation*}
    i_k(t)=G_k\cdot u(t)=G_k\cdot \frac{i(t)}{G_{eq}} \ \ \rightsquigarrow \ \  \boxed{i_k(t)=\frac{G_k}{\sum^n_{j=1} G_j}\cdot i(t)}
\end{equation*}

Es consecuencia inmediata de que todas las resistencias están sometidas a la misma tensión.
\end{block}
\end{frame}

\section[]{Circuitos equivalentes}

\begin{frame}
\frametitle{Circuitos equivalentes}
\begin{block}{Objetivo}
	\justifying
Simplificar parte de un circuito del cual no interesa su estudio, reemplazando varios componentes por un número más reducido, con la condición de que las respuestas en el resto del circuito no queden afectadas por dicha sustitución.
\end{block}


\begin{figure}[hbt]
\epsfig{file=./figuras_tema_2/circ_equivalentes.eps,height=3cm}
\label{circ_equivalentes}
\end{figure}

Para que ambas situaciones tenga la misma solución:
\begin{equation*}
f_o[u(t),i(t)]=f_e[u(t),i(t)] \ \ \Rightarrow \ \  \boxed{  u_e(t)=u_o(t)\ \ ; \ \  i_e(t)=i_o(t)}
\end{equation*}


\end{frame}


\section[]{Regla de sustitución}


\begin{frame}
\frametitle{Regla de sustitución}

\begin{small}
Si se reemplaza un circuito monopuerta por una fuente ideal de tensión (intensidad), cuyo valor sea el de la tensión (intensidad) en la puerta, entonces la intensidad (tensión) de la puerta, y por tanto el resto de respuestas en el circuito exterior no se alteran, como consecuencia de la unicidad de la solución en circuitos lineales.
\end{small}

\vspace{-0.2cm}
\begin{figure}[hbt]
\epsfig{file=./figuras_tema_2/regla_sustitucion.eps,height=3.8cm}
\label{regla_sustitucion}
\end{figure}


\vspace{-0.4cm}

\begin{small}
Las fuentes utilizadas en la regla de sustitución dependen de la topología y de los componentes de ambos circuitos. Por lo tanto,  \textbf{no constituyen un equivalente del circuito}.
\end{small}

\end{frame}
