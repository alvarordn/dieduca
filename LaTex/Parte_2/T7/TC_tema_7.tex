\section{Contexto y justificación del uso de corriente alterna sinusoidal}

\begin{frame}
\frametitle{Contexto histórico}

\begin{itemize}
	\justifying
	\item El uso de corriente alterna sinusoidal está ligado a las máquinas eléctricas para generación y transporte de energía.
	\item \textbf{1832}. Primer generador de corriente eléctrica de Hippolyte Pixii.
	\item \textbf{Guerra de las corrientes}. Finales de la década de 1880:
	\begin{itemize}
		\item Thomas A. Edison, \textit{General Electric}. Corriente continua (DC).
		\item Nikola Tesla, \textit{Westinghouse Electric}. Corriente alterna (AC).
	\end{itemize}
	\item Se impuso la corriente alterna, usada mayoritariamente hoy en día.
	\item División entre empresas europeas (50 Hz) y americanas (60 Hz).
\end{itemize}

\begin{frame}
\frametitle{Ondas sinusoidales}


\begin{exampleblock}{Ejemplo de tensión sinusoidal}

Determinar la frecuencia angular, fase inicial, frecuencia, valor de pico y valor eficaz de la siguiente tensión sinusoidal:
 \begin{equation*}
u(t)= 70\,\sqrt{2}\,\cos(50\,t+60^\circ)\,\text{V}
 \end{equation*}


\begin{small}
\textbf{Solución}: $\omega =50$ rad/s; $\varphi=60^\circ$; $f=\frac{50}{2\,\pi}$\,Hz; $U_{p}=70\,\sqrt{2} $ V; $U=70$ V
\end{small}

\end{exampleblock}

\begin{exampleblock}{Ejemplo de intensidad sinusoidal}

Determinar la frecuencia angular, fase inicial, frecuencia, valor de pico  y valor eficaz de la siguiente intensidad sinusoidal:
 \begin{equation*}
i(t)=20\,\sin(\pi\,1000\,t+30^\circ)\,\text{A}
 \end{equation*}

\begin{small}
	\textbf{Solución}: $\omega =3141.6$ rad/s; $\varphi=-60^\circ$; $f=500$ Hz; $I_p = 20$ A; $I =\frac{20}{\sqrt{2}}$ A  
\end{small}


\end{exampleblock}

\section{Concepto de fasor}
\begin{frame}
\frametitle{Concepto de fasor}

\justifying
El \textbf{fasor} es un número complejo asociado a una señal sinusoidal, cuyo módulo es el valor eficaz y su fase la fase inicial de la señal.
\vspace{-0.5cm}
\begin{columns}
\begin{column}[t]{6cm}
\begin{center}
\structure{Representación temporal}
\end{center}
\end{column}
\begin{column}[t]{6cm}
\begin{center}
\structure{Representación Fasorial}
\end{center}
\end{column}
\end{columns}
\vspace{-0.3cm}
\begin{columns}
\begin{column}{6cm}
\begin{equation*}
\begin{split}
m(t)&=\sqrt{2}\cdot M\cdot \cos(\omega t+\varphi)\\
&=\Re e \{\sqrt{2}\cdot M e^{j\left(\omega t+\varphi\right)}\}\\
&=\Re e \{\sqrt{2}\cdot e^{j\omega t}\cdot {\underbrace{M e^{j\varphi}}_{Fasor}} \}
\end{split}
\end{equation*}
\end{column}
\begin{column}{6cm}
\begin{center}
$\underbrace{\mathcal{M}\triangleq M e^{j\varphi}= M\angle \varphi}$\\
$\Downarrow$\\
Fasor asociado  a $m(t)$
\end{center}
\end{column}
\end{columns}

\begin{block}{Impedancia compleja [Ohmio, $\Omega$]}

\begin{columns}
\begin{column}{5.5cm}

    \mathcal{Z}=\frac{{U}\angle \varphi_u}{{I}\angle \varphi_i}=Z\angle \varphi=R+jX
\end{equation*}

$$\varphi = \varphi_u - \varphi_i$$
\begin{equation*}
\begin{split}
R \  &\equiv \ \text{Resistencia}\\ %%
%X \  &\equiv \ \text{Reactancia} 
X>0 \  &\equiv \ \text{Reactancia inductiva}\\%%
X<0 \  &\equiv \ \text{Reactancia capacitiva}
\end{split}
\end{equation*}

\section{Análisis de circuitos en régimen permanente sinusoidal}


\begin{frame}
\frametitle{Leyes de Kirchhoff en corriente alterna. Intensidades}

\begin{equation*}
\begin{split}
    \sum_{k} i_k(t)&=\sum_{k} \sqrt{2}I_k\cos(\omega t+\varphi_k)\\
    &=\sum_{k} \Re e [\sqrt{2}e^{j\omega t} I_k  e^{j\varphi_k}]\\
    &=\sum_{k} \Re e [\sqrt{2}e^{j\omega t} \mathcal{I}_k]=0 \ \ \Leftrightarrow \ \ \boxed{\sum_{k}\mathcal{I}_k=0}
\end{split}
\end{equation*}
\begin{figure}[hbt]
\centering
\epsfig{file=./figuras_tema_7/1K_AC.eps,height=2cm}
\end{figure}

\end{frame}


\begin{frame}

\vspace{-0.3cm}

 \sum_{k} u_k(t)&=\sum_{k} \sqrt{2}U_k\cos(\omega t+\varphi_k)\\

    &=\sum_{k} \Re e [\sqrt{2}e^{j\omega t} \mathcal{U}_k]=0 \ \ \Leftrightarrow \ \ \boxed{\sum_{k}\mathcal{U}_k=0}
    \end{split}
 \end{equation*}

\begin{enumerate}
  \item Elementos de la matriz \textbf{Y}$_N$
\begin{equation*}
\begin{split}
\mathcal{Y}_{N_{(j,j)}}&=\sum_{k}\mathcal{Y}_{jk} \ \ \text{sumatorio de admitancias conectadas al nudo $j$}\\
\mathcal{Y}_{N_{(j,k)}}&=-\mathcal{Y}_{jk} \ \ \text{admitancia total entre el nudo $j$ y el nudo $k$}
\end{split}
\end{equation*}

  \item Elementos del vector \textbf{I}$_N$
    \begin{equation*}
\begin{array}{cl}
  \mathcal{I}_{N_j}=\sum_{j} \mathcal{I}_{N_{pj}}-\mathcal{I}_{N_{nj}} &   \text{sumatorio de las fuentes de intensidad}\\
   &  \text{que inciden en el nudo $j$, positivas si entran}\\
   &  \text{en el nudo y negativas si salen}
\end{array}
 \end{equation*}
\end{enumerate}
\end{small}

\end{block}
\end{frame}

    \right)}
\end{equation*}
\vspace{-0.3cm}
\begin{enumerate}
  \item Elementos de la matriz \textbf{Z}$_M$
\begin{equation*}
\begin{split}
\mathcal{Z}_{M_{(j,j)}}&=\sum_{k}\mathcal{Z}_{jk} \ \ \text{sumatorio de impedancias pertenecientes a la malla $j$}\\
\mathcal{Z}_{M_{(j,k)}}&=-\mathcal{Z}_{jk} \ \ \text{impedancia total común de la malla $j$ y la malla $k$}
\end{split}
\end{equation*}
  \item Elementos del vector \textbf{U}$_M$
    \begin{equation*}
\begin{array}{cl}
\mathcal{U}_{M_j}=\sum_{j} \mathcal{U}_{M_{pj}}-\mathcal{U}_{M_{nj}} &   \text{sumatorio de las fuentes de tensión que}\\
   &  \text{pertenecen a la malla $j$, con signo positivo si }\\
   & \text{la intensidad de malla sale por el terminal} \\
   & \text{positivo de la fuente, y negativo si entra}
   \end{array}
 \end{equation*}
\end{enumerate}
\end{small}

\end{block}
\end{frame}
