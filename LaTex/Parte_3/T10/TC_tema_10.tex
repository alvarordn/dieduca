u_2(t)&=\sqrt{2}U_F\cos(\omega t-120^\circ)\\

\begin{equation*}
    p(t)=\underbrace{u_1(t)\cdot i_1(t)}_{p_1(t)}+\underbrace{u_2(t)\cdot i_2(t)}_{p_2(t)}+\underbrace{u_3(t)\cdot i_3(t)}_{p_3(t)}=3U_FI_F\cos \varphi
\end{equation*}

\end{frame}






\begin{frame}{Potencia instantánea}
\vspace{-0.5cm}
\begin{equation*}
  \boxed{p(t)=3U_FI_F\cos \varphi}
\end{equation*}

\structure{Observaciones:}

\begin{itemize}
	\justifying
\item La potencia instantánea total de un sistema trifásico equilibrado es constante, no apareciendo términos fluctuantes. Esto conlleva beneficios como disminución de vibraciones, ruido y fatiga en las máquinas eléctricas rotativas.

\item Si el sistema no está equilibrado, la potencia instantánea contendrá en general términos fluctuantes.

\item Si bien los términos fluctuantes se cancelan entre sí, ello no evita que exista un intercambio de potencia reactiva entre cada una de las fases y el circuito exterior (puede existir un desfase entre la tensión e intensidad de cada fase). 

\end{itemize}



\end{frame}



\section{Cálculo de potencia a través de magnitudes de fase y de línea}



\begin{frame}{Potencia activa trifásica}
En términos de las magnitudes de fase:
\begin{equation*}
\boxed{P=\frac{1}{T}\int^{T}_0 p(t) dt=3 U_F I_F\cos \varphi}
\end{equation*}

En términos de las magnitudes de línea:
\begin{equation*}
\left.
    \begin{split}
    \text{Conexión en estrella:} & \ U_F=\frac{U_L}{\sqrt{3}} \ ; \ I_F=I_L\\
    \text{Conexión en triángulo:} & \ U_F=U_L \ ; \ I_F=\frac{I_L}{\sqrt{3}}
    \end{split}\right\} \ \ \Rightarrow  \ \ \boxed{P=\sqrt{3} U_L I_L\cos \varphi}
\end{equation*}

\justifying
El ángulo $\varphi$ es el desfase entre las correspondientes tensiones e intensidades de fase (en un sistema equilibrado, las tres fases tienen el mismo desfase).

\end{frame}


\begin{frame}{Potencia reactiva trifásica}
	\justifying
Definida como la suma de las potencias reactivas en cada fase. En términos de las magnitudes de fase:
\begin{equation*}
\boxed{Q=3 U_F I_F\sin \varphi}
\end{equation*}

En términos de las magnitudes de línea:
\begin{equation*}
\left.
    \begin{split}
   \text{Conexión en estrella:} & \ U_F=\frac{U_L}{\sqrt{3}} \ ; \ I_F=I_L\\
    \text{Conexión en triángulo:} & \ U_F=U_L \ ; \ I_F=\frac{I_L}{\sqrt{3}}
    \end{split}\right\} \ \ \Rightarrow  \ \ \boxed{Q=\sqrt{3} U_L I_L\sin \varphi}
\end{equation*}

\justifying
El ángulo $\varphi$ es el desfase entre las correspondientes tensiones e intensidades de fase (en un sistema equilibrado, las tres fases tienen el mismo desfase).


\end{frame}



\begin{frame}{Potencia compleja y aparente en sistemas trifásicos}
\structure{Potencia compleja:} tiene una expresión análoga a la vista para circuitos monofásicos.
\begin{equation*}
\boxed{\mathcal{S}=3\cdot \mathcal{S}_F=3\cdot \mathcal{U}_{F}\mathcal{I}^*_{F}=\sqrt{3}\cdot \mathcal{U}_{L}\mathcal{I}^*_{L}=P+jQ }
\end{equation*}

\justifying
\structure{Potencia aparente:} se corresponde con el módulo de la potencia compleja.
\begin{equation*}
\boxed{{S}=|\mathcal{S}|=3\cdot {U}_{F}{I}_F=\sqrt{3}\cdot {U}_{L} {I}_L=\sqrt{P^2+Q^2}}
\end{equation*}

\end{frame}

	\end{exampleblock}
\end{frame}


\section{Mejora del factor de potencia}

\begin{frame}
\frametitle{Definición de factor de potencia}
\justifying
Al igual que en circuitos monofásicos, el \textbf{factor de potencia} se define como el cociente entre la potencia activa y la potencia aparente:
\begin{equation*}
    \boxed{\text{fdp}=\frac{P}{S}=\frac{P}{\sqrt{P^2+Q^2}}=\frac{\sqrt{3}\cdot U_L I_L\cos \varphi}{\sqrt{3}\cdot U_L I_L}=\cos \varphi}
\end{equation*}

\begin{alertblock}{}
El factor de potencia \textbf{NO} da información sobre el signo de la potencia reactiva. Es necesario indicar el carácter inductivo o capacitivo.
\end{alertblock}


\end{frame}
