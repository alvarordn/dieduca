\section{Transformador ideal}

\begin{frame}{Transformador ideal}

\justifying
Bobinas {\bf perfectamente acopladas} que comparten flujo magnético a través de un núcleo de material ferromagnético de {\bf reluctancia nula}

\begin{enumerate}
\item Acoplamiento perfecto
\vspace*{0.2cm}

\begin{columns}
\begin{column}{5cm}
%\begin{figure}[hbt]
\centering
\epsfig{file=./figuras_tema_11/bobi_acop_nucleo_A_perfec.eps,height=2.5cm}
%\end{figure}
\end{column}
\hspace{0.3cm}
\begin{column}{3cm}
\begin{equation*}
 \phi_1(t)=\phi_2(t)
% =\phi_{12}(t)+\phi_{21}(t)
\end{equation*}
\begin{equation*}
  {\begin{split}
    u_1(t)&=N_1\,\frac{d\phi_1}{dt}
% =L_1\cdot \frac{d i_1}{dt}+M\cdot  \frac{d i_2}{dt}
    \\
  u_2(t)&=N_2\,\frac{d\phi_2}{dt}
%   =M\cdot \frac{d i_1}{dt}+L_2 \cdot \frac{d i_2}{dt}
    \end{split}}
\end{equation*}
\end{column}
\end{columns}
Por tanto, $u_1(t)=n\cdot u_2(t)$, con $n=N_1/N_2$

\item Reluctancia nula, $\mathcal{R}\simeq 0$
$$ \mathcal{F} = N_1\cdot i_1(t) + N_2\cdot i_2(t) = \mathcal{R} \cdot \phi(t) \simeq 0$$
Por tanto, $i_1(t)=-\frac{1}{n}\, i_2(t)$
\end{enumerate}

\end{frame}

\begin{frame}{Adaptación de elementos pasivos}

%\vspace{-0.2cm}

\begin{figure}[h]
 \centering
 \epsfig{file=./figuras_tema_11/adaptacion.eps,height=2.0cm}
  \end{figure}

\vspace{-0.5cm}

\begin{equation*}
    u_1(t)=n\, u_2(t)=n\, f[i_2(t)]=n\, f[ni_1(t)]
\end{equation*}
donde $f(\cdot)$ es la función que define la característica $u-i$ del circuito lineal conectado en el secundario del transformador:
\begin{equation*}
    u_1(t)=n\, f[n\, i_1(t)]=n^2\, f[i_1(t)]
\end{equation*}

\begin{alertblock}{}
	\justifying
El conjunto formado por un transformador ideal y un elemento pasivo conectado en su secundario equivale desde el primario a un elemento de la misma naturaleza.
Por ejemplo:
\vspace{-0.4cm}
\begin{equation*}
    R_1=n^2\,R_2 \ \ ; \ \ L_1=n^2\,L_2 \ \ ; \ \ C_1=\frac{1}{n^2}\,C_2
\end{equation*}

\end{alertblock}
\end{frame}

\section{Principio de funcionamiento de las máquinas eléctricas rotativas}

\begin{frame}{Principio de funcionamiento}

\begin{columns}
\begin{column}{6cm}
\begin{figure}[h]
 \centering
 \epsfig{file=./figuras_tema_11/alternador_elemental.eps,height=3cm}
  \end{figure}
\end{column}
\begin{column}{6cm}
\begin{small}
\begin{itemize}
	\justifying
  \item La bobina 1 (\textbf{rotor}) se alimenta con corriente continua
  \item Parte del núcleo magnético, que soporta a la bobina 1, se separa mecánicamente del resto del núcleo y se la hace girar con velocidad $\omega$
   \end{itemize}
  \end{small}
\end{column}
\end{columns}

\begin{small}
\begin{itemize}
   \item El flujo constante creado por la bobina  1 es visto como un flujo variable por la bobina 2 (\textbf{estátor}) debido al coeficiente de inducción mutua ($m(t)$)
  \end{itemize}
\end{small}
\begin{equation*}
    u_2(t)=N_2\cdot \frac{d \phi(t)}{dt}=\frac{d}{dt}\left[m(t) I_1  \right]=I_1\cdot \frac{d m(t)}{dt}
\end{equation*}
\justifying
Con un diseño constructivo adecuado se puede conseguir que $m(t)$ sea aproximadamente sinusoidal, y en consecuencia el flujo $\phi(t)$ también lo será. \textbf{Por tanto $u_2(t)$ será una magnitud sinusoidal}

\end{frame}

\begin{frame}{Mediante devanados polifásicos. Teorema de Ferraris}
	\vspace{-0.2cm}
	\begin{block}{Devanado trifásico de $1$ par de polos}
	\begin{columns}
	\begin{column}{6cm}
	\vspace{-0.7cm}
	\begin{figure}[h]
	\centering
	  \epsfig{file=./figuras_tema_11/m3_p1.eps,height=3.7cm}
	 \end{figure}

	\begin{column}{6cm}
	\begin{small}
	\begin{equation*}
	    \mathcal{F}_r(\alpha,t)=\mathcal{F}_R(\alpha,t)+\mathcal{F}_S(\alpha,t)+\mathcal{F}_T(\alpha,t)
	\end{equation*}
	\vspace{-0.4cm}
	\begin{equation*}
	    \begin{split}
	    \mathcal{F}_R(\alpha,t)&=F_{m,R}(t)\cdot \sin(\alpha)\\
	    \mathcal{F}_S(\alpha,t)&=F_{m,S}(t)\cdot \sin(\alpha-120^\circ)\\
	    \mathcal{F}_T(\alpha,t)&=F_{m,T}(t)\cdot \sin(\alpha+120^\circ)
	    \end{split}
	\end{equation*}
	\end{small}
	Según el teorema de Ampere:
	\vspace{-0.2cm}
	\begin{small}
	\begin{equation*}
	    F_{m,k}(t)\propto i_k(t)
	\end{equation*}
	\end{small}
	\end{column}
	\end{columns}
	\vspace{0.2cm}
	Si por las bobinas circula un sistema trifásico equilibrado de intensidades:
	\begin{small}
	\begin{equation*}
	\left.
	    \begin{split}
	    \mathcal{F}_R&=F_m \cdot \sin(\omega t)\cdot \sin(\alpha)\\
	    \mathcal{F}_S&=F_m \cdot \sin(\omega t-120^\circ)\cdot \sin(\alpha-120^\circ)\\
	    \mathcal{F}_T&=F_m \cdot \sin(\omega t+120^\circ)\cdot \sin(\alpha+120^\circ)\\
	    \end{split}
	    \right\} \  \Rightarrow \  \boxed{\mathcal{F}_r(\alpha,t)=\frac{3}{2}F_m\cdot \sin(\alpha-\omega t)}
	\end{equation*}
	\end{small}
	\vspace{-0.3cm}
	\end{block}
\end{frame}



\begin{frame}{Mediante devanados polifásicos. Teorema de Ferraris}
	
	\begin{block}{Devanado trifásico de $p$ pares de polos}
		\justifying
	Si el devanado trifásico tiene $p$ pares de polos entonces la velocidad de giro del campo magnético giratorio (denominada \textbf{velocidad de sincronismo}) es:
	\begin{equation*}
	    n_e=\frac{60\cdot f}{p} \ \ [\text{rpm}]
	\end{equation*}
	donde $f$ es la frecuencia de las corrientes de alimentación y $p$ es el número de pares de polos
	
	%Se puede observar que dicha velocidad es función de la frecuencia y del número de pares de polos de la máquina.
	\end{block}
	
	\vspace{-0.2cm}

\end{frame}
